#!/bin/bash
# Launch pmac6200 from a known state.
#
# A MAME machine's state is NOT just the disk image. It is:
#   disks/pmac6200-ata.hd      the hard disk
#   nvram/pmac6200/cuda        PRAM: startup disk, monitor depth, sound volume
#   cfg/pmac6200.cfg           MAME config: monitor type config port, inputs
#
# Restoring only the disk leaves PRAM/cfg from a later run and the MkLinux
# Booter then fails with "cannot find bootstrap file". Always restore or clear
# all three together.
#
# Usage:
#   ./run-pmac6200.sh fresh           restore disk from snapshot, clear PRAM+cfg
#   ./run-pmac6200.sh keep            restore disk from snapshot, keep PRAM+cfg
#   ./run-pmac6200.sh snapshot NAME   save disk+PRAM+cfg together as NAME
#   ./run-pmac6200.sh restore NAME    restore all three from NAME
#   ./run-pmac6200.sh macos           boot the Mac OS CD instead of MkLinux
set -u

SNAP=disks/pmac6200-MKLINUX.hd
DISK=disks/pmac6200-ata.hd
CD="MkLinux R2 RC5.toast"
MACOS_CD="MAC_OS_7-6-1_RETAIL.ISO"
STATE=disks/state

clone() { cp -c "$1" "$2" 2>/dev/null || cp "$1" "$2"; }

stop() {
	pkill -f cordyceps 2>/dev/null
	for _ in 1 2 3 4 5; do
		pgrep -f cordyceps >/dev/null || break
		sleep 1
	done
	pkill -9 -f cordyceps 2>/dev/null
	sleep 2
	if pgrep -f cordyceps >/dev/null; then
		echo "FATAL: could not stop existing instance(s):" >&2
		pgrep -lf cordyceps >&2
		exit 1
	fi
}

case "${1:-fresh}" in
snapshot)
	name="${2:?need a name}"; stop
	mkdir -p "$STATE/$name"
	clone "$DISK" "$STATE/$name/disk.hd"
	cp nvram/pmac6200/cuda "$STATE/$name/cuda" 2>/dev/null
	cp cfg/pmac6200.cfg "$STATE/$name/cfg" 2>/dev/null
	echo "saved disk+PRAM+cfg as '$name'"; exit 0 ;;
restore)
	name="${2:?need a name}"; stop
	rm -f "$DISK"; clone "$STATE/$name/disk.hd" "$DISK"
	mkdir -p nvram/pmac6200
	cp "$STATE/$name/cuda" nvram/pmac6200/cuda 2>/dev/null || rm -f nvram/pmac6200/cuda
	cp "$STATE/$name/cfg" cfg/pmac6200.cfg 2>/dev/null || rm -f cfg/pmac6200.cfg
	echo "restored all three from '$name'" ;;
fresh)
	stop
	rm -f "$DISK"; clone "$SNAP" "$DISK"
	rm -f nvram/pmac6200/cuda cfg/pmac6200.cfg
	echo "disk restored from snapshot; PRAM and cfg cleared" ;;
keep)
	stop
	rm -f "$DISK"; clone "$SNAP" "$DISK"
	echo "disk restored from snapshot; PRAM and cfg kept" ;;
macos)
	stop; CD="$MACOS_CD"
	echo "booting Mac OS CD" ;;
*)
	echo "unknown mode: $1"; exit 1 ;;
esac

echo "disk md5(first 512M): $(dd if="$DISK" bs=1M count=512 2>/dev/null | md5 -q)"
nohup ./cordyceps pmac6200 -window -ram 64m -hard "$DISK" -cdrom "$CD" \
	> /tmp/pmac6200.log 2>&1 &
sleep 8
n=$(pgrep -f cordyceps | wc -l | tr -d ' ')
if [ "$n" -ne 1 ]; then
	echo "FATAL: expected exactly 1 instance, found $n" >&2
	pgrep -lf cordyceps >&2
	tail -5 /tmp/pmac6200.log >&2
	exit 1
fi
echo "running: 1 instance (pid $(pgrep -f cordyceps))"
exit 0
