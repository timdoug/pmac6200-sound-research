#!/usr/bin/env python3
"""Summarise a pmac6200 chime recording: per-channel level and a time profile."""

import struct
import sys
import wave


def summarise(path):
    with wave.open(path, "rb") as w:
        nch = w.getnchannels()
        width = w.getsampwidth()
        rate = w.getframerate()
        n = w.getnframes()
        raw = w.readframes(n)

    if width != 2:
        print(f"{path}: unexpected sample width {width}")
        return

    flat = struct.unpack("<%dh" % (len(raw) // 2), raw)
    chans = [flat[c::nch] for c in range(nch)]

    print(f"{path}")
    print(f"  {n / rate:.1f}s @ {rate} Hz, {nch} channels")

    for c, s in enumerate(chans):
        lo, hi = min(s), max(s)
        dc = sum(s) / len(s)
        # AC energy about the channel's own mean, which ignores a DC offset
        ac = sum(abs(v - dc) for v in s) / len(s)
        print(f"  ch{c}: min {lo:7d}  max {hi:7d}  dc {dc:9.1f}  mean|ac| {ac:8.2f}")

    # per-second AC energy of the loudest channel, to locate any transient
    best = max(range(nch), key=lambda c: max(chans[c]) - min(chans[c]))
    s = chans[best]
    dc = sum(s) / len(s)
    print(f"  per-second mean|ac| on ch{best}:")
    line = []
    for sec in range(int(n / rate)):
        blk = s[sec * rate:(sec + 1) * rate]
        if not blk:
            break
        line.append(f"{sum(abs(v - dc) for v in blk) / len(blk):.0f}")
    print("    " + " ".join(line))


for p in sys.argv[1:]:
    summarise(p)
